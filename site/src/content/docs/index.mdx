## Read a repository

Open a repository, resolve `HEAD` and count the commits reachable from it. The
[getting started](/girt/getting-started/) page explains each step.

```rust title="src/main.rs"
use girt::refs::RefName;
use girt::{HistoryLimits, PackLimits, Repository};

let repository = Repository::open("path/to/repo")?;
let head = repository.references()?.resolve(&RefName::new("HEAD")?, 5)?;
let objects = repository.objects(PackLimits::default())?;
if let Some(id) = head.id {
    // Walk ancestry without trusting commit timestamps.
    let count = objects.walk(&[id], HistoryLimits::default())?.len();
    println!("{count} commits reachable from HEAD");
}
```
