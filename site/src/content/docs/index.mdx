import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components';

<CardGrid>
  <Card title="Git's concepts, directly" icon="seti:rust">
    Objects, trees, commits, references, the index and repositories are Rust types. Writing an
    object, moving a reference and updating a working tree are separate calls.
  </Card>
  <Card title="Git formats, exactly" icon="approve-check">
    SHA-1 and SHA-256 repositories, loose and packed objects, files and reftable references, and
    index v2–v4. Every format girt supports is checked against the Git executable.
  </Card>
  <Card title="Explicit refusals" icon="warning">
    Unsupported inputs return an error instead of a guess. Operations that delete or rewrite data
    verify the replacement before they retire anything.
  </Card>
  <Card title="Built for jj" icon="rocket">
    girt's first consumer is [jj](https://github.com/jj-vcs/jj). An experimental jj branch runs on
    girt with no gix, libgit2 or Git subprocess.
  </Card>
</CardGrid>
