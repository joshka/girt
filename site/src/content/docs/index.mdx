## Read a repository

Print the first ten commits in `HEAD`'s history. The [getting started](/girt/getting-started/) page
builds this program step by step.

```rust title="src/main.rs"
fn main() -> Result<(), Box<dyn std::error::Error>> {
    let repository = girt::Repository::open(".")?;

    let head = repository.head()?;
    let head_id = head.id.ok_or("HEAD has no commits yet")?;

    let objects = repository.objects()?;
    let history = objects.walk(&[head_id])?;

    for id in history.into_iter().take(10) {
        let commit = objects.read_commit(id)?;
        let message = String::from_utf8_lossy(commit.message());
        let summary = message.lines().next().unwrap_or_default();
        println!("{id} {summary}");
    }

    Ok(())
}
```
